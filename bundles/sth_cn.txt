#type unit

单位空间计算 "hitSize"/8 如96/8=12
单位组装厂生产空间计算：第一个单位 "hitSize"*0.225 如96*0.225=21.6

#type turret
"reload": x,    重新装填时间(数值越低越快)
"range": x,    炮台感知范围(与炮台子弹射程独立，以x*8为一格，如6*8=48[既6格要感知范围要填48])
"maxAmmo": x, 最大弹药容量，如果ammoMultiplier太大，而maxAmmo太小可能不能装填弹药?(理论上)
"recoilTime": x, 开炮制退时间
"recoil": x,      开炮制退距离(每8位一格?[没测])
"cooldownTime": x,  冷却时间(不知道有什么用)
"shoot": {           对炮弹&炮管编辑
		"type": "ShootBarrel",    子弹类型
		"shots": 6,               一次射出多少子弹
		"shotDelay": 2,          射击延迟(前摇)
		"barrels": [             炮管编辑(炮管x.y.°[x坐标，y坐标，角度])
			3.5, 13, 0,
			-3.5, 13, 0,
			10.5, 13, 0,
			-10.5, 13, 0
		]
	},
"collidesGround": 炮台是否对地面单位进行感知用true/false(与targetGround独立下文介绍)
"collidesAir":      炮台是否对地面单位进行感知用true/false(与targetAir独立下文介绍)



#type bullet
"targetAir":      填true/false子弹是否对空中单位碰撞(对空中目标造成伤害)
"targetGround": 填true/false子弹是否对地面单位碰撞(对地面目标造成伤害)
"fromColor": "F595BE",       子弹颜色        
"toColor": "F595BE",          子弹颜色(如果可以渐变)
"ammoMultiplier": 4,           弹药装填倍率，越高一个物品可以填充炮台子弹量越高，如果ammoMultiplier太大，而maxAmmo太小可能不能装填弹药?(理论上)
"shootEffect": "thoriumShoot",   射击效果(不知道有什么用)
"smokeEffect": "thoriumShoot"  射击烟雾(不知道有什么用)
"type": "ShrapnelBulletType",    子弹类型
"reloadMultiplier": 1.2,           射速倍率
"width": 20,                    子弹宽度(如果子弹用的是图片的话)
"length": 160,                  子弹长度(如果子弹用的是图片的话)
"damage": 125,                子弹伤害
"splashDamageRadius": 15,    范围伤害范围(每8为一格子)
